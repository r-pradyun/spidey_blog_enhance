#!/usr/bin/env python3 """ Firebase Security Scanner - Complete Automation Extracts Firebase credentials from decompiled APK and runs security scan Usage: python firebase_scanner.py Example: python firebase_scanner.py /path/to/decompiled/app """ import subprocess import yaml import re import sys import os import time import requests from pathlib import Path from typing import List, Set class Colors: """ANSI color codes for terminal output""" RED = '\033[0;31m' GREEN = '\033[0;32m' YELLOW = '\033[1;33m' BLUE = '\033[0;34m' MAGENTA = '\033[0;35m' CYAN = '\033[0;36m' BOLD = '\033[1m' NC = '\033[0m' # No Color class FirebaseScanner: """Extract Firebase credentials from decompiled APK using Nuclei templates""" def __init__(self, apk_dir: str, templates_dir: str): self.apk_dir = Path(apk_dir) self.templates_dir = Path(templates_dir) self.project_ids: Set[str] = set() self.api_keys: Set[str] = set() self.firebase_domains: Set[str] = set() def run_nuclei_template(self, template_path: str) -> str: """Run a single nuclei template and return raw output""" print(f"{Colors.CYAN}[*] Running nuclei template: {template_path.split('/')[-1]}{Colors.NC}") cmd = [ "nuclei", "-t", template_path, "-target", str(self.apk_dir), "-file", "-silent" ] try: result = subprocess.run( cmd, capture_output=True, text=True, check=False ) return result.stdout + result.stderr except FileNotFoundError: print(f"{Colors.RED}[!] Error: nuclei command not found. Please install nuclei.{Colors.NC}") sys.exit(1) except Exception as e: print(f"{Colors.RED}[!] Error running nuclei: {e}{Colors.NC}") return "" def extract_project_id_from_domain(self, domain: str) -> str: """Extract Firebase project ID from domain""" patterns = [ r'([a-z0-9._-]+)\.appspot\.com', r'([a-z0-9._-]+)\.firebaseio\.com', r'([a-z0-9._-]+)\.firebaseapp\.com' ] for pattern in patterns: match = re.search(pattern, domain, re.IGNORECASE) if match: return match.group(1) return None def parse_nuclei_results(self, raw_output: str, template_type: str): """Parse nuclei raw output with regex""" if not raw_output: return if template_type == 'api-key': api_key_pattern = r'AIza[0-9A-Za-z\-_]{35}' matches = re.findall(api_key_pattern, raw_output) for match in matches: self.api_keys.add(match) print(f"{Colors.GREEN}[+] Found API Key: {match}{Colors.NC}") elif template_type == 'appspot': domain_pattern = r'[a-z0-9._-]+\.appspot\.com' matches = re.findall(domain_pattern, raw_output, re.IGNORECASE) for match in matches: self.firebase_domains.add(match) project_id = self.extract_project_id_from_domain(match) if project_id: self.project_ids.add(project_id) print(f"{Colors.GREEN}[+] Found Firebase domain: {match} -> Project ID: {project_id}{Colors.NC}") elif template_type == 'firebase-db': patterns = [ r'[a-z0-9._-]+\.firebaseio\.com', r'[a-z0-9._-]+\.firebaseapp\.com' ] for pattern in patterns: matches = re.findall(pattern, raw_output, re.IGNORECASE) for match in matches: self.firebase_domains.add(match) project_id = self.extract_project_id_from_domain(match) if project_id: self.project_ids.add(project_id) print(f"{Colors.GREEN}[+] Found Firebase domain: {match} -> Project ID: {project_id}{Colors.NC}") def scan_with_nuclei(self): """Run all nuclei templates and extract credentials""" print(f"\n{Colors.YELLOW}[*] Scanning target: {self.apk_dir}{Colors.NC}") print(f"{Colors.YELLOW}[*] Running nuclei templates...{Colors.NC}\n") templates = { 'api-key': 'Google-api.yaml', 'appspot': 'appspot.yaml', 'firebase-db': 'firebase-database.yaml' } for template_type, template_name in templates.items(): template_path = self.templates_dir / template_name if not template_path.exists(): print(f"{Colors.RED}[!] Warning: Template not found: {template_name}{Colors.NC}") continue raw_output = self.run_nuclei_template(str(template_path)) self.parse_nuclei_results(raw_output, template_type) print(f"\n{Colors.BOLD}[*] Scan complete!{Colors.NC}") print(f"{Colors.MAGENTA}[*] Found {len(self.project_ids)} unique project ID(s){Colors.NC}") print(f"{Colors.MAGENTA}[*] Found {len(self.api_keys)} unique API key(s){Colors.NC}") print(f"{Colors.MAGENTA}[*] Found {len(self.firebase_domains)} unique Firebase domain(s){Colors.NC}") def generate_config(self, output_file: str = "config.yaml") -> bool: """Generate firescan configuration file""" if not self.project_ids: print(f"{Colors.RED}[!] No Firebase project IDs found. Cannot generate config.{Colors.NC}") return False if not self.api_keys: print(f"{Colors.RED}[!] No API keys found. Cannot generate config.{Colors.NC}") return False project_id = list(self.project_ids)[0] api_key = list(self.api_keys)[0] config = { 'projectId': project_id, 'apiKey': api_key } with open(output_file, 'w') as f: yaml.dump(config, f, default_flow_style=False) print(f"\n{Colors.GREEN}[+] Generated config file: {output_file}{Colors.NC}") print(f"{Colors.CYAN} Project ID: {project_id}{Colors.NC}") print(f"{Colors.CYAN} API Key: {api_key[:20]}...{Colors.NC}") return True class FirescanAutomation: """Automate firescan execution""" def __init__(self, config_file: str, output_dir: str, project_id: str): self.config_file = config_file self.output_dir = Path(output_dir) self.project_id = project_id # Get the firescan path relative to this script script_dir = Path(__file__).parent self.firescan_path = Path("/usr/bin/firescan") def run_firescan_commands(self, commands: List[str]): """Run firescan with a list of commands""" print(f"\n{Colors.YELLOW}[*] Running firescan commands...{Colors.NC}") print(f"{Colors.CYAN}[*] Commands: {', '.join(commands)}{Colors.NC}") # Prepare commands cmd_input = "\n".join(commands) + "\n" try: # Check if firescan exists if not self.firescan_path.exists(): print(f"{Colors.RED}[!] Error: firescan not found at {self.firescan_path}{Colors.NC}") return # Run firescan with config - output findings.json to the input directory findings_path = self.output_dir / "findings.json" result = subprocess.run( [str(self.firescan_path), "--json-output", str(findings_path), "--config", self.config_file], input=cmd_input, text=True, timeout=600 # 10 minute timeout ) print(f"{Colors.GREEN}[+] Firescan completed{Colors.NC}") except subprocess.TimeoutExpired: print(f"{Colors.RED}[!] Firescan timed out after 10 minutes{Colors.NC}") except FileNotFoundError: print(f"{Colors.RED}[!] Error: firescan command not found{Colors.NC}") except Exception as e: print(f"{Colors.RED}[!] Error running firescan: {e}{Colors.NC}") def wait_for_email_verification(self, timeout: int = 300, poll_interval: int = 5) -> bool: """Wait for email verification via webhook""" webhook_url = f"https://n8n.payatu.sh/webhook-test/firebase-mailverification?proj={self.project_id}" print(f"\n{Colors.YELLOW}[*] Waiting for email verification...{Colors.NC}") print(f"{Colors.CYAN}[*] Webhook: {webhook_url}{Colors.NC}") start_time = time.time() while time.time() - start_time < timeout: try: response = requests.get(webhook_url, timeout=10) if response.status_code == 200: data = response.json() if data.get("status") == "success": print(f"{Colors.GREEN}[+] Email verification successful!{Colors.NC}") return True print(f"{Colors.CYAN}[*] Waiting... (elapsed: {int(time.time() - start_time)}s){Colors.NC}") time.sleep(poll_interval) except requests.exceptions.RequestException as e: print(f"{Colors.RED}[!] Request error: {e}{Colors.NC}") time.sleep(poll_interval) except Exception as e: print(f"{Colors.RED}[!] Error: {e}{Colors.NC}") time.sleep(poll_interval) print(f"{Colors.RED}[!] Email verification timed out after {timeout}s{Colors.NC}") return False def run_comprehensive_scan(self): """Run essential firescan commands""" print(f"\n{Colors.BLUE}{'='*60}{Colors.NC}") print(f"{Colors.BOLD}[*] Starting Firebase security scan{Colors.NC}") print(f"{Colors.BLUE}{'='*60}{Colors.NC}") # Phase 1: Create account phase1_commands = [ "show options", "auth --create-account -e payatu.user1@gmail.com -P TestPass123", "save-quit" ] print(f"\n{Colors.YELLOW}[*] Phase 1 - Creating account:{Colors.NC}") for i, cmd in enumerate(phase1_commands, 1): print(f"{Colors.CYAN} {i}. {cmd}{Colors.NC}") self.run_firescan_commands(phase1_commands) # Wait for email verification if not self.wait_for_email_verification(): print(f"{Colors.RED}[!] Email verification failed. Aborting scan.{Colors.NC}") return # Phase 2: Continue with remaining commands phase2_commands = [ "show options", "auth --enum-providers", "scan --all", "save-quit" ] print(f"\n{Colors.YELLOW}[*] Phase 2 - Running security scan:{Colors.NC}") for i, cmd in enumerate(phase2_commands, 1): print(f"{Colors.CYAN} {i}. {cmd}{Colors.NC}") self.run_firescan_commands(phase2_commands) print(f"\n{Colors.GREEN}[+] Scan complete!{Colors.NC}") def print_summary(config_file: str, findings_path: str): """Print final summary""" print(f"\n{Colors.BOLD}Outputs:{Colors.NC}") print(f"{Colors.CYAN} • {config_file} (Generated configuration){Colors.NC}") print(f"{Colors.CYAN} • {findings_path} (Comprehensive scan results){Colors.NC}") print(f"\n{Colors.BLUE}{'='*60}{Colors.NC}\n") def main(): """Main execution function""" # Check arguments if len(sys.argv) < 2: print(f"{Colors.RED}Usage: python {sys.argv[0]} {Colors.NC}") print(f"\n{Colors.YELLOW}Example:{Colors.NC}") print(f" python {sys.argv[0]} /path/to/decompiled/app") sys.exit(1) apk_path = sys.argv[1] # Default templates directory (relative to script location) script_dir = Path(__file__).parent templates_dir = script_dir / "templates" config_file = "config.yaml" # Validate paths if not os.path.exists(apk_path): print(f"{Colors.RED}[!] Error: Target directory not found: {apk_path}{Colors.NC}") sys.exit(1) if not os.path.exists(templates_dir): print(f"{Colors.RED}[!] Error: Templates directory not found: {templates_dir}{Colors.NC}") sys.exit(1) # Print banner print(f"{Colors.GREEN}Target: {Colors.NC}{apk_path}") print(f"{Colors.GREEN}Templates: {Colors.NC}{templates_dir}") print() # Step 1: Extract credentials print(f"{Colors.YELLOW}{'━'*60}{Colors.NC}") print(f"{Colors.BOLD}[STEP 1] Extracting Firebase credentials...{Colors.NC}") print(f"{Colors.YELLOW}{'━'*60}{Colors.NC}") scanner = FirebaseScanner(apk_path, str(templates_dir)) scanner.scan_with_nuclei() # Step 2: Generate config print(f"\n{Colors.YELLOW}{'━'*60}{Colors.NC}") print(f"{Colors.BOLD}[STEP 2] Generating firescan configuration...{Colors.NC}") print(f"{Colors.YELLOW}{'━'*60}{Colors.NC}") if not scanner.generate_config(config_file): print(f"{Colors.RED}[!] Failed to generate config. Exiting.{Colors.NC}") sys.exit(1) # Step 3: Run comprehensive firescan print(f"\n{Colors.YELLOW}{'━'*60}{Colors.NC}") print(f"{Colors.BOLD}[STEP 3] Running comprehensive security scan...{Colors.NC}") print(f"{Colors.YELLOW}{'━'*60}{Colors.NC}") # Get project_id from scanner project_id = list(scanner.project_ids)[0] if scanner.project_ids else "" automation = FirescanAutomation(config_file, apk_path, project_id) automation.run_comprehensive_scan() # Print summary findings_path = str(Path(apk_path) / "findings.json") print_summary(config_file, findings_path) if __name__ == "__main__": main()
